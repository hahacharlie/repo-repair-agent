1. Install system dependencies (requires root):
apt-get update && apt-get install -y libavformat-dev libavcodec-dev libavutil-dev libswresample-dev libsdl3-dev libavdevice-dev libdrm-dev libusb-1.0-0-dev

2. Make gradlew executable (no root needed):
chmod +x gradlew

3. Build the project:
sh -c "set -e; if [ ! -f scrcpy-server ]; then if command -v wget >/dev/null 2>&1; then wget -q -O scrcpy-server https://github.com/Genymobile/scrcpy/releases/download/v5.0.1/scrcpy-server-v5.0.1; else curl -fsSL -o scrcpy-server https://github.com/Genymobile/scrcpy/releases/download/v5.0.1/scrcpy-server-v5.0.1; fi; fi; printf \"%s  %s\\n\" \"764eb6f79811d5211fe9df341120882ba9994c7a61b897d7bf3fb662e53bc536\" \"scrcpy-server\" | sha256sum --check; if [ -f build/build.ninja ]; then meson setup build --reconfigure -Dprebuilt_server=scrcpy-server; else meson setup build -Dprebuilt_server=scrcpy-server; fi && meson compile -C build"

4. Run tests:
sh -c 'meson test -C build'
