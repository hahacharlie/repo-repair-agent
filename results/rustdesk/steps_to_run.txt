Run these environment setup commands in order. Commands marked (root) require root privileges.

git clone --depth 1 https://github.com/rustdesk/hbb_common.git libs/hbb_common (no root)

apt-get update && apt-get install -y libx11-dev libxcb-randr0-dev libxdo-dev libxfixes-dev libxcb-shape0-dev libxcb-xfixes0-dev libasound2-dev libpulse-dev libgtk-3-dev clang libclang-dev cmake ninja-build libgstreamer1.0-dev libgstreamer-plugins-base1.0-dev pkg-config nasm yasm git curl wget zip unzip make gcc g++ (root)

apt-get update && apt-get install -y libssl-dev (root)

printf "export CXXFLAGS='-include cstdint'\n" >> /home/repair/env (no root)

apt-get update && apt-get install -y libvpx-dev libyuv-dev libaom-dev libopus-dev (root)

cat > /usr/lib/x86_64-linux-gnu/pkgconfig/libyuv.pc <<'EOF'
prefix=/usr
includedir=${prefix}/include
libdir=${prefix}/lib/x86_64-linux-gnu

Name: libyuv
Description: Library for YUV scaling
Version: 0.0.1922
Libs: -L${libdir} -lyuv
Cflags: -I${includedir}
EOF
chmod 644 /usr/lib/x86_64-linux-gnu/pkgconfig/libyuv.pc
pkg-config --exists libyuv && echo 'libyuv.pc installed' || echo 'libyuv.pc missing' (root)

Build the project:
cargo build --features linux-pkg-config

Run tests:
cargo test --locked --release --lib --features linux-pkg-config
